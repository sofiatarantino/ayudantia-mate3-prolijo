\begin{definition} \textbf{Punto aislado.} \label{def:pto_ais}
    Sean $A \subseteq \Rn{n}$ y $\vect{x}_0 \in A$. Decimos que $\vect{x}_0$ es un \emph{punto aislado} de $A$ si $\vect{x}_0 \in A$ y no es punto de acumulación de $A$. Es decir, si:
    \[
     \exists \delta > 0 \colon B^*_{\delta}(\vect{x}_0) \cap A = \varnothing. 
    \]
\end{definition}
