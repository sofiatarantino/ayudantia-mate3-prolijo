\begin{definition} \textbf{Punto de acumulación.} \label{def:pto_acum}
    Sean $A \subseteq \Rn{n}$ y $\vect{x}_0 \in \Rn{n}$. Decimos que $\vect{x}_0$ es un \emph{punto de acumulación} de $A$ si
    \[
     \forall \delta > 0 \colon B^*_{\delta}(\vect{x}_0) \cap A \neq \varnothing. 
    \]
    Denotamos al conjunto de todos los puntos de acumulación de $A$ con $A'$:
    \[
     A' = \left\lbrace \vect{x} \in \Rn{n} \colon \vect{x} \text{ es punto de acumulación de } A \right\rbrace.
    \]
\end{definition}
