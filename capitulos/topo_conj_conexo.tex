\begin{definition} \textbf{Conjunto conexo.} \label{def:conexo}
    Sea $A \subseteq \Rn{n}$. Decimos que $A$ es un conjunto \emph{conexo} si no existen dos conjuntos abiertos $U, V \subseteq \Rn{n}$ disjuntos y no vacíos tales que cumplan simultáneamente:
    \[
     A \cap U \neq \varnothing, \quad A \cap V \neq \varnothing \quad \text{y} \quad A \subseteq U \cup V.
    \]
\end{definition}
